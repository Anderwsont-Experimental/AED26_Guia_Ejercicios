/*
Por su gran desempeño en la cátedra de Algoritmos y Estructuras de Datos, MercadoPago lo
ha contratado como pasante para desempeñar una importante tarea dentro de la empresa.

Como primera actividad, se le ha solicitado desarrollar un algoritmo para actualizar los
saldos de las cuentas de los usuarios. Para esto, se cuenta con los siguientes archivos:

    CUENTAS, archivo secuencial, ordenado por nro_usuario
    nro_usuario | apellido | nombre | cvu | cuil | monto | fecha_nacimiento | fecha_alta  | estado(V o F)

y un archivo de movimientos

    MOVIMIENTOS
    nro_usuario | cuil | cod_movimiento | importe

Donde el cod_movimiento puede ser:
    1 – Alta de un nuevo usuario
    2 – Transferencia, el usuario ha realizado una transferencia de dinero
    3 – Ingreso, el usuario ha recibido un ingreso de dinero
    4 – Baja de la cuenta, la cuenta del usuario será desactivada (cambia el estado a Falso)

En caso de que se trate del alta de un nuevo usuario, los datos personales del usuario serán
capturados de una archivo del RENAPER, el cual se encuentra indexado por el campo cuil

    RENAPER, indexado por cuil
    cuil | nombre | apellido | fecha_nacimiento | direccion

Se le solicita además, que al finalizar, se informe:
    • Número de nuevas cuentas
    • Porcentaje de cuentas que se dieron de baja


Para generar la fecha_alta, considerar la existencia de una funcion HOY() que devuelve como
dato un registro con la fecha actual.

Como segunda actividad, además de lo anterior, se le ha solicitado que, durante el proceso,
mantenga actualizado el archivo indexado de USUARIOS, con los nuevos datos ingresados:

    USUARIOS, indexado por cuil
    cuil  | nro_usuario  | nombre | apellido | fecha_nacimiento | direccion

*/

ACCION Nombre_Accion ES
    Ambiente
        ctas_tot,ctas_new,ctas_baja:entero
        //definicion de registros y archivos
        reg_fecha = Registro
            aa: 1900..2050
            mm: 1..12
            dd: 1..31
        Fin_Registro

        reg_cuentas = Registro
            nro_usuario: N(12)
            apellido: AN(35)
            nombre: AN(35)
            cvu: N(22)
            cuil: N(11)
            monto: N(10,2)
            fecha_nacimiento: reg_fecha
            fecha_alta: reg_fecha
            estado: logico
        Fin_Registro
        arch_cuentas, arch_act_cuentas: archivo de reg_cuentas ordenado por nro_usuario
        cuenta, aux_cuenta: reg_cuentas 

        reg_movimientos = Registro
            nro_usuario: N(12)
            cuil: N(11)
            cod_movimiento: (1,2,3,4) //1:alta|2:transf|3:ingreso|4:baja logica
            importe: N(10,2)
        Fin_Registro
        arch_movimientos: archivo de reg_movimientos ordenado por nro_usuario y cod_mov
        mov: reg_movimientos

        reg_renaper = Registro
            cuil: N(11)
            nombre: AN(35)
            apellido: AN(35)
            fecha_nacimiento: reg_fecha
            direccion: AN(51)
        Fin_Registro
        arch_renaper: archivo de reg_renaper indexado por cuil
        dato_renaper: reg_renaper

        //definicion de subacciones

        PROCEDIMIENTO leer_arch_cuentas ES //maestro
            LEER(arch_cuentas, cuenta)
            SI FDA(arch_cuentas) ENTONCES
                cuenta.clave := high_value
            FIN_SI
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO leer_arch_movimientos ES //movimiento
                LEER(arch_movimientos, mov)
                SI FDA(arch_movimientos) ENTONCES
                    mov.clave := high_value
                FIN_SI
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO CARGA_alta_cuentas ES
            aux_cuenta.nro_usuario:= mov.nro_usuario
            aux_cuenta.apellido:= dato_renaper.apellido
            aux_cuenta.nombre:= dato_renaper.nombre
            aux_cuenta.cvu:= NULL
            aux_cuenta.cuil:= mov.cuil
            aux_cuenta.monto:= mov.importe
            aux_cuenta.fecha_nacimiento:= dato_renaper.fecha_nacimiento
            aux_cuenta.fecha_alta:= HOY()
            aux_cuenta.estado:= VERDADERO
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO PROCESO_evalua_movimiento ES
            SEGUN mov.cod_movimiento HACER
                2:  //transferencia (se resta importe)
                    aux_cuenta.monto := aux_cuenta.monto + mov.importe
                3: //ingreso (se suma importe)
                    aux_cuenta.monto := aux_cuenta.monto - mov.importe
                4: //baja
                    ctas_baja:= ctas_baja+1 //si se dio de alta y baja en el mismo mov, cuenta en alta tmb
                    aux_cuenta.estado := FALSO
                otros : ESCRIBIR("error en codigo de movimiento")
            FIN_SEGUN
        FIN_PROCEDIMIENTO

    Proceso
        //INICIALIZACION DE ARCHIVOS
        ABRIR E/(arch_cuentas);ABRIR E/(arch_movimientos);ABRIR E/(arch_renaper)
        leer_arch_cuentas
        leer_arch_movimientos
        ABRIR /S(arch_act_cuentas)
        ctas_tot:= 0
        ctas_new:= 0
        ctas_baja:= 0

        MIENTRAS (cuenta.nro_usuario <> HV) O (mov.nro_usuario <> HV) HACER
            ctas_tot:= ctas_tot+1
            SI (cuenta.nro_usuario < mov.nro_usuario) ENTONCES //maestro sin movimiento
                ESCRIBIR(arch_act_cuentas,cuenta)
                leer_arch_cuentas
            SINO
                SI (cuenta.nro_usuario = mov.nro_usuario) ENTONCES //maestro con mov
                    aux_cuenta:= cuenta
                    MIENTRAS (cuenta.nro_usuario = mov.nro_usuario) HACER
                        PROCESO_evalua_movimiento
                        leer_arch_movimientos
                    FIN_MIENTRAS
                SINO //alta
                    ctas_new:= ctas_new+1
                    dato_renaper.cuil := mov.cuil
                    LEER(arch_renaper,dato_renaper)
                    SI (EXISTE) ENTONCES
                        CARGA_alta_cuentas
                        leer_arch_movimientos
                        MIENTRAS (cuenta.nro_usuario = mov.nro_usuario) HACER
                            PROCESO_evalua_movimiento
                            leer_arch_movimientos
                        FIN_MIENTRAS
                        ESCRIBIR(arch_act_cuentas,aux_cuenta)
                    SINO
                        ESCRIBIR("ALTA DE CUIL NO EXISTENTE EN RENAPER")
                    FIN_SI
                FIN_SI
            FIN_SI
        FIN_MIENTRAS
        ESCRIBIR("altas:",ctas_new,"% bajas:",ctas_baja/ctas_tot*100)

FIN_ACCION
