        Resg2: N(4)

        //Definicion de Procedimientos ADAPTABLES al enunciado

        //Definicion de procedimientos estandarizados. Ajustar a cada consigna.
            /*NOTA: Para escribir archivos de salida, usamos el nivel de
                jerarquia de corte menor EQUIVALENTE al menor campo en el
                archivo de salida que se pide.
                En los cortes de nivel jerarquia mayor no se vuelve a escribir el archivo de salida
                Utilizar variables de RESGUARDO para la carga
                Si arch_sal tiene acumulacion por cont2, Esc(arch_sal,x) se hace en corte_2*/

        //Jerarquia MENOR
        PROCEDIMIENTO corte_1 es
            Esc("Para el ", Resg1, " existen: ", cont1) //Emitir resultados del nivel

            //Escribir archivo de salida (Ver NOTA)

            Resg1 := r.clave1 //Resguardar la nueva clave de este nivel
        FIN_PROCEDIMIENTO

        //Jerarquia INTERMEDIA
        PROCEDIMIENTO corte_2 es
            corte_1 //llamada al corte_n-1
            Esc("Para el ", Resg2, " existen: ", cont2) //Emitir resultados del nivel
            cont3 := cont3 + cont2 //Acumular totales al nivel superior
            //Escribir archivo de salida (Ver NOTA)
            cont2 := 0 //Reiniciar totales de este nivel
            Resg2 := r.clave2 //Resguardar la nueva clave de este nivel
        FIN_PROCEDIMIENTO

        //Jerarquia MAYOR
        PROCEDIMIENTO corte_3 es
            corte_2 //llamada al corte_n-1
            Esc("Para el ", Resg3, " existen: ", cont3) //Emitir resultados del nivel
            conT := contT + cont3 //Acumular totales al nivel superior
            //Escribir archivo de salida (Ver NOTA)
            cont3 := 0 //Reiniciar totales de este nivel
            Resg3 := r.clave3 //Resguardar la nueva clave de este nivel
        FIN_PROCEDIMIENTO
