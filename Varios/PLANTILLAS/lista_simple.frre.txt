/*PLANTILLA de funciones y procedimientos generales de listas simplemente enlazadas
P: puntero de recorrido
Q: puntero de generacion de nodo
AUX: puntero auxiliar, normalmente usado para guardar posicion anterior a P
*/

ACCION plantilla_busqueda(prim: puntero a nodo) ES
    Ambiente
        //definicion de registro de nodo
        nodo = Registro
            dato: tipo_dato
            prox: puntero a nodo
        Fin_Registro
        //def de punteros
        P: puntero a nodo
        //variables
        valor: tipo_dato

    Proceso
        //determinacion del valor buscado
        LEER(valor)
        //inicializacion del puntero de recorrido "P"
        P:= prim
        //recorrido de la lista
        MIENTRAS (P <> NIL) Y (*p.dato <> valor) HACER //2do parentesis debe ser reemplazado por condicion de consigna
            P := *P.prox
        FIN_MIENTRAS
        //evaluacion del resultado del recorrido
        SI (P = NIL) ENTONCES
            ESCRIBIR("No se encontro el elemento")
        SINO
            ESCRIBIR("se encontro el elemento")
            // "P" apunta al nodo que contiene el elemento buscado
        FIN_SI
FIN_ACCION

ACCION plantilla_CARGA_apilada() ES //NO ES UNA PILA
    Ambiente
        //definicion de registro de nodo
        nodo = Registro
            dato: tipo_dato
            prox: puntero a nodo
        Fin_Registro
        //def de punteros
        prim, Q: puntero a nodo
        //variables
        valor: tipo_dato

    Proceso
        //Toma del valor a cargar
        LEER(valor)
        //inicializacion de prim
        prim := NIL
        MIENTRAS (valor <> "*") HACER //"*" es marca de fin de carga
            //generacion en memoria de nuevo nodo
            NUEVO(Q)
            //carga al inicio de lista
            *Q.dato := valor
            *Q.prox := prim
            prim := Q
            //toma de nuevo valor
            LEER(valor)
        FIN_MIENTRAS
FIN_ACCION

ACCION plantilla_CARGA_encolada() ES //NO ES UNA COLA
    Ambiente
        //definicion de registro de nodo
        nodo = Registro
            dato: tipo_dato
            prox: puntero a nodo
        Fin_Registro
        //def de punteros
        prim, Q, AUX: puntero a nodo
        //variables
        valor: tipo_dato

    Proceso
        //determinacion del valor a cargar
        LEER(valor)
        //inicializacion de prim
        prim := NIL
        MIENTRAS (valor <> "*") HACER //"*" es marca de fin de carga
            //generacion en memoria de nuevo nodo
            NUEVO(Q)
            //carga al final de lista
            *Q.dato := valor
            *Q.prox := nil
            SI (prim = nil) ENTONCES //lista vacia
                prim := Q
            SINO //lista con elementos
                *AUX.prox := Q
            FIN_SI
            // Resguardo el puntero Q
            AUX := Q
            //toma de nuevo valor
            LEER(valor)
        FIN_MIENTRAS
FIN_ACCION

ACCION plantilla_CARGA_ordenada() ES //Desde vacio
    Ambiente
        //definicion de registro de nodo
        nodo = Registro
            dato: tipo_dato
            prox: puntero a nodo
        Fin_Registro
        //def de punteros
        prim, P, Q, AUX: puntero a nodo
        //variables
        valor: tipo_dato

    Proceso
        //determinacion del valor a cargar
        LEER(valor)
        //inicializacion de prim
        prim := NIL
        MIENTRAS (valor <> "*") HACER //"*" es marca de fin de carga
            //creacin del nuevo nodo
            NUEVO(Q)
            *Q.dato := valor
            //Busqueda de posicion de insercion
            P := prim
            AUX := NIL
            MIENTRAS (P <> NIL) Y (*P.dato < valor) HACER //2do parentesis debe ser reemplazado por condicion de consigna
                AUX := P
                P := *P.prox
            FIN_MIENTRAS
            //evaluacion de la busqueda
            SI (AUX = NIL) ENTONCES //caso 1 y 2
                *Q.prox := prim
                prim := Q
                //puntero P queda apuntando al nodo posterior a Q
            SINO //caso 3 y 4
                *Q.prox := P
                *AUX.prox := Q
                //puntero P queda con dir de nodo posterior a Q
                //puntero AUX queda con dir de nodo previo a Q
            FIN_SI
            //puntero Q queda apuntando al nodo recien insertado.
            //toma de nuevo valor
            LEER(valor)
        FIN_MIENTRAS
FIN_ACCION

ACCION plantilla_eliminacion(prim: puntero a nodo) ES
    Ambiente
        //definicion de registro de nodo
        nodo = Registro
            dato: tipo_dato
            prox: puntero a nodo
        Fin_Registro
        //def de punteros
        P, AUX: puntero a nodo
        //variables
        valor: tipo_dato
    Proceso
        //determinacion del valor a cargar
        LEER(valor)
        //Busqueda de nodo a eliminar
        P := prim
        AUX := NIL
        MIENTRAS (P <> NIL) Y (*P.dato <> valor) HACER //2do parentesis debe ser reemplazado por condicion de consigna
            AUX := P
            P := *P.prox
        FIN_MIENTRAS
        //evaluacion de la busqueda
        SI (P = nil) ENTONCES
            ESCRIBIR("Error, no se encuentra el valor")
        SINO
            SI (AUX = nil) ENTONCES //caso 1 y 2
                prim := *P.prox
            SINO //caso 3 y 4
                *AUX.prox := *P.prox
                //puntero AUX queda con posicion previa a nodo a eliminar
            FIN_SI
            //Disponibilizacion de la memoria ocupada por nodo.
            DISPONER(P)
        FIN_SI
FIN_ACCION
