ACCION actualizacion_lotes ES
    Ambiente
        //definicion de registros y archivos
        formato_clave = Registro
            clave1: ...
            clave2: ...
            ...
            claveN: ...
        Fin_Registro

        formato_maestro = Registro
            clave: formato_clave
            campo1: ...
            campo2: ...
            ...
            campoN: ...
            marca_baja: ... //Reg especial de MAESTRO
        Fin_Registro

        formato_movimiento = Registro
            clave: formato_clave
            campo1: ...
            campo2: ...
            ...
            campoN: ...
            cod_mov: ("A","M","B") //Reg especial de MOVIMIENTO
        Fin_Registro

        arch_mae, arch_mae_act: archivo de formato_maestro ordenado por clave
        reg_mae, reg_act, reg_aux: formato_maestro

        arch_mov: archivo de formato_movimiento ordenado por clave y cod_mov
        reg_mov: formato_movimiento

        //definicion de subacciones

        PROCEDIMIENTO leer_maestro ES//
            LEER(arch_mae, reg_mae)
            SI FDA(arch_mae) ENTONCES
                reg_mae.clave := high_value
            FIN_SI
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO leer_movimiento ES//
                LEER(arch_mov, reg_mov)
                SI FDA(arch_mov) ENTONCES
                    reg_mov.clave := high_value
                FIN_SI
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO CARGA_alta_maestro ES
                reg_aux.clave := reg_mov.clave
                reg_aux.campo1 := reg_mov.campo1
                reg_aux.campo2 := reg_mov.campo2
                ...
                reg_aux.campoN := reg_mov.campoN
                reg_aux.marca_baja := "" //estado inicial "activo"
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO CARGA_modif_maestro ES
            //no se actualizan los campos claves.
            SI (reg_mov.campo1 <> "") ENTONCES
                reg_aux.campo1 := reg_mov.campo1
            FIN_SI
            SI (reg_mov.campo2 <> "") ENTONCES
                reg_aux.campo2 := reg_mov.campo2
            FIN_SI
            SI (reg_mov.campoN <> "") ENTONCES
                reg_aux.campoN := reg_mov.campoN
            FIN_SI
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO Baja_Registro(tipo:("F","L")) ES //F: baja fisica - L: Baja logica
            SI (tipo = 'L') ENTONCES // ## baja LOGICA
                //En vez de *, se puede asignar la fecha del dia, o alterar otro campo, segun el problema.
                reg_aux.marca_baja := "*"
            SINO
                SI (tipo = 'F') ENTONCES // ## baja FISICA
                    //acciones a realizar durante baja fisica, segun consigna.(Resguardos, lista de autores, etc)
                    //Si no hay acciones, dejar en blanco, para saltar el registro.
                FIN_SI
            FIN_SI
        FIN_PROCEDIMIENTO

        PROCEDIMIENTO PROCESO_evalua_movimiento ES
            //movimientos se cargan en reg_aux
            SI (reg_mov.cod_mov = "A") ENTONCES
                ESCRIBIR("Error en alta. Clave ya existe")
                reg_aux := reg_mae //se pasa maestro sin cambios.
            SINO
                SI (reg_mov.cod_mov = "M") ENTONCES
                    //modificacion
                    CARGA_modif_maestro
                SINO
                    SI (reg_mov.cod_mov = "B") ENTONCES
                        //baja logica o fisica.
                        Baja_Registro("L/F") //El argumento puede ser también una variable definida por usuario.
                    FIN_SI
                FIN_SI
            FIN_SI
        FIN_PROCEDIMIENTO

    Proceso
        //INICIALIZACION DE ARCHIVOS
        ABRIR E/(arch_mae); ABRIR E/(arch_mov)
        leer_maestro
        leer_movimiento
        ABRIR /S(arch_mae_act)

        //PROCESAMIENTO ARCHIVOS
        MIENTRAS (reg_mae.clave <> high_value) O (reg_mov.clave <> high_value) HACER
            SI (reg_mae.clave < reg_mov.clave) ENTONCES // ## MAESTRO SIN Movimiento
                reg_act := reg_mae
                ESCRIBIR(arch_mae_act, reg_act)
                leer_maestro

            SINO
                SI (reg_mae.clave = reg_mov.clave) ENTONCES // ## MAESTRO CON Movimiento
                    reg_aux := reg_mae
                    //barrido de lote de movimientos:
                    MIENTRAS (reg_mae.clave = reg_mov.clave) HACER
                        PROCESO_evalua_movimiento
                        leer_movimiento
                    FIN_MIENTRAS
                    reg_act := reg_aux
                    ESCRIBIR(arch_mae_act, reg_act)
                    leer_maestro

                SINO // ## Movimiento SIN MAESTRO
                    /*Movimienntos posibles:
                        > 1 alta (si o si)
                        > 0 a n modificaciones
                        > 0 a 1 baja */
                    CARGA_alta_maestro
                    leer_movimiento
                    //CORREGIR TEMA DE CARGA DE AUX
                    //barrido de lotes de movimientos restantes, si existen
                    MIENTRAS (clave_reg_aux = clave_mov) HACER
                        PROCESO_evalua_movimiento
                        leer_movimiento
                    FIN_MIENTRAS
                    //escritura maestro actualizado con movimientos
                    reg_act := reg_aux
                    ESCRIBIR(arch_mae_act, reg_act)
                FIN_SI
            FIN_SI
        FIN_MIENTRAS

        //ENTREGA RESULTADOS

        //CIERRE DE ARCHIVOS
        CERRAR(arch_mae)
        CERRAR(arch_mae_act)
        CERRAR(arch_mov)

FIN_ACCION
